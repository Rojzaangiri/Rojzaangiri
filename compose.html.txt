<script>
  const STORAGE_KEY = 'noticesData';  // Using same notices storage for messages

  const form = document.getElementById('composeForm');
  const successMsg = document.getElementById('successMsg');

  form.addEventListener('submit', e => {
    e.preventDefault();

    const title = document.getElementById('msgTitle').value.trim();
    const message = document.getElementById('msgBody').value.trim();
    const rolesSelect = document.getElementById('recipientRoles');
    const roles = Array.from(rolesSelect.selectedOptions).map(opt => opt.value);
    const date = new Date().toLocaleDateString();

    if (!title || !message || roles.length === 0) {
      alert('Please fill all fields and select at least one recipient.');
      return;
    }

    const allMessages = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];

    const newMessage = {
      title,
      message,
      date,
      roles
    };

    allMessages.unshift(newMessage);
    localStorage.setItem(STORAGE_KEY, JSON.stringify(allMessages));

    form.reset();
    successMsg.style.display = 'block';

    setTimeout(() => {
      successMsg.style.display = 'none';
    }, 3000);
  });
</script>
