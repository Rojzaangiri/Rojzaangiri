    <label for="newSubject">Add New Subject</label>
    <input type="text" id="newSubject" placeholder="Enter subject name" required />
    <button type="submit">Add Subject</button>
  </form>

<script>
  const subjectsTableBody = document.querySelector('#subjectsTable tbody');
  const addSubjectForm = document.getElementById('addSubjectForm');
  const newSubjectInput = document.getElementById('newSubject');

  // Sample initial subjects (can be empty or loaded from backend)
  let subjects = [
    "Mathematics",
    "English",
    "Science",
    "Nepali",
    "Social Studies",
    "Computer",
    "Art",
  ];

  // Function to render the subjects table
  function renderSubjects() {
    subjectsTableBody.innerHTML = '';
    subjects.forEach((subject, index) => {
      const row = document.createElement('tr');

      // Subject name cell with inline edit capability
      const nameCell = document.createElement('td');
      const nameInput = document.createElement('input');
      nameInput.type = 'text';
      nameInput.value = subject;
      nameInput.className = 'editable';
      nameInput.dataset.index = index;
      nameInput.addEventListener('change', (e) => {
        const i = e.target.dataset.index;
        subjects[i] = e.target.value.trim() || subjects[i];
        renderSubjects();
      });
      nameCell.appendChild(nameInput);

      // Actions cell with delete button
      const actionsCell = document.createElement('td');
      actionsCell.className = 'actions';
      const deleteBtn = document.createElement('button');
      deleteBtn.textContent = 'Delete';
      deleteBtn.addEventListener('click', () => {
        if(confirm(`Delete subject "${subject}"?`)) {
          subjects.splice(index, 1);
          renderSubjects();
        }
      });
      actionsCell.appendChild(deleteBtn);

      row.appendChild(nameCell);
      row.appendChild(actionsCell);

      subjectsTableBody.appendChild(row);
    });
  }

  // Add subject form submit handler
  addSubjectForm.addEventListener('submit', e => {
    e.preventDefault();
    const newSubj = newSubjectInput.value.trim();
    if(newSubj && !subjects.includes(newSubj)) {
      subjects.push(newSubj);
      newSubjectInput.value = '';
      renderSubjects();
    } else {
      alert('Please enter a unique subject name.');
    }
  });

  // Initial render
  renderSubjects();
</script>
