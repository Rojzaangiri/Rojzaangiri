<script>
  const STORAGE_KEY = 'feeLedgerData';

  let ledgerData = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];

  const feeForm = document.getElementById('fee-form');
  const ledgerBody = document.getElementById('ledger-body');
  const filterStudent = document.getElementById('filter-student');

  // Save ledger data to localStorage
  function saveLedger() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(ledgerData));
  }

  // Render ledger table rows
  function renderLedger(filterByStudent = '') {
    ledgerBody.innerHTML = '';

    let filteredData = ledgerData;
    if (filterByStudent) {
      filteredData = ledgerData.filter(record => record.studentId === filterByStudent);
    }

    if (filteredData.length === 0) {
      ledgerBody.innerHTML = `<tr><td colspan="6" class="no-data">No payment records found.</td></tr>`;
      return;
    }

    filteredData.forEach((record, index) => {
      const tr = document.createElement('tr');

      tr.innerHTML = `
        <td>${record.studentName}</td>
        <td>${record.studentId}</td>
        <td>${record.paymentDate}</td>
        <td>Rs. ${record.amount.toFixed(2)}</td>
        <td>${record.paymentMode}</td>
        <td>
          <button class="action-btn" onclick="deleteRecord(${index})">Delete</button>
        </td>
      `;

      ledgerBody.appendChild(tr);
    });
  }

  // Populate filter dropdown with unique student IDs and names
  function populateFilter() {
    // Clear old options except first
    filterStudent.innerHTML = '<option value="">-- All Students --</option>';

    const uniqueStudents = {};
    ledgerData.forEach(record => {
      if (!uniqueStudents[record.studentId]) {
        uniqueStudents[record.studentId] = record.studentName;
      }
    });

    for (const id in uniqueStudents) {
      const option = document.createElement('option');
      option.value = id;
      option.textContent = `${uniqueStudents[id]} (ID: ${id})`;
      filterStudent.appendChild(option);
    }
  }

  // Delete a payment record
  function deleteRecord(index) {
    if (confirm('Are you sure you want to delete this record?')) {
      ledgerData.splice(index, 1);
      saveLedger();
      populateFilter();
      renderLedger(filterStudent.value);
    }
  }

  // Form submit handler
  feeForm.addEventListener('submit', e => {
    e.preventDefault();

    const studentName = document.getElementById('student-name').value.trim();
    const studentId = document.getElementById('student-id').value.trim();
    const paymentDate = document.getElementById('payment-date').value;
    const amount = parseFloat(document.getElementById('amount').value);
    const paymentMode = document.getElementById('payment-mode').value;

    if (!studentName || !studentId || !paymentDate || !amount || !paymentMode) {
      alert('Please fill all fields.');
      return;
    }

    const newRecord = {
      studentName,
      studentId,
      paymentDate,
      amount,
      paymentMode
    };

    ledgerData.push(newRecord);
    saveLedger();

    feeForm.reset();
    populateFilter();
    renderLedger(filterStudent.value);
  });

  // Filter change handler
  filterStudent.addEventListener('change', () => {
    renderLedger(filterStudent.value);
  });

  // Initial load
  populateFilter();
  renderLedger();

  // Make deleteRecord globally accessible (for inline onclick)
  window.deleteRecord = deleteRecord;
</script>
