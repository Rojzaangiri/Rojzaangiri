// Helper to format date YYYY-MM-DD
function formatDate(d) {
return d.toISOString().split('T')[0];
}

// Generate array of dates between two dates inclusive (max 365 days)
function dateRange(start, end) {
const dates = [];
const curr = new Date(start);
const last = new Date(end);
while(curr <= last) {
dates.push(new Date(curr));
curr.setDate(curr.getDate() + 1);
if(dates.length > 365) break; // safety cap
}
return dates;
}

// Calculate attendance percentage for person over given dates
function calculatePresentPercent(personId, dates) {
if(dates.length === 0) return 'N/A';
let presentCount = 0;
let totalCount = dates.length;
for(const d of dates) {
const key = ${personId}_${formatDate(d)};
if(attendanceRecords[key] === 'present') {
presentCount++;
}
}
return ((presentCount / totalCount) * 100).toFixed(1) + '%';
}
