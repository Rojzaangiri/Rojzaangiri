<script>
  const STORAGE_KEY = 'questionBankFiles';
  const fileInput = document.getElementById('file-input');
  const fileList = document.getElementById('file-list');
  const clearBtn = document.getElementById('clear-btn');

  // Load saved files from localStorage or empty array
  let filesData = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];

  function saveToStorage() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(filesData));
  }

  function renderFiles() {
    fileList.innerHTML = '';
    if (filesData.length === 0) {
      clearBtn.style.display = 'none';
      return;
    }
    clearBtn.style.display = 'inline-block';

    filesData.forEach((fileObj, index) => {
      const div = document.createElement('div');
      div.classList.add('file-item');

      const removeBtn = document.createElement('button');
      removeBtn.classList.add('remove-btn');
      removeBtn.innerText = '×';
      removeBtn.title = 'Remove file';
      removeBtn.onclick = () => {
        filesData.splice(index, 1);
        saveToStorage();
        renderFiles();
      };

      div.appendChild(removeBtn);

      if (fileObj.type.startsWith('image/')) {
        const img = document.createElement('img');
        img.src = fileObj.dataUrl;
        img.alt = fileObj.name;
        div.appendChild(img);

        const name = document.createElement('div');
        name.className = 'file-name';
        name.textContent = fileObj.name;
        div.appendChild(name);
      } else if (fileObj.type === 'application/pdf') {
        const pdfIcon = document.createElement('div');
        pdfIcon.innerHTML = '📄';
        pdfIcon.style.fontSize = '48px';
        div.appendChild(pdfIcon);

        const link = document.createElement('a');
        link.href = fileObj.dataUrl;
        link.download = fileObj.name;
        link.textContent = fileObj.name;
        link.className = 'pdf-link';
        div.appendChild(link);
      }

      fileList.appendChild(div);
    });
  }

  fileInput.addEventListener('change', e => {
    const files = Array.from(e.target.files);

    files.forEach(file => {
      if (!['application/pdf', 'image/png', 'image/jpeg'].includes(file.type)) {
        alert('Only PDF, PNG, and JPEG files are allowed.');
        return;
      }

      const reader = new FileReader();
      reader.onload = (ev) => {
        // Check file size - localStorage max ~5MB, limit per file to 2MB here
        if (file.size > 2 * 1024 * 1024) {
          alert(`File ${file.name} is too large (max 2MB). Skipped.`);
          return;
        }

        filesData.push({
          name: file.name,
          type: file.type,
          dataUrl: ev.target.result
        });
        saveToStorage();
        renderFiles();
      };

      reader.readAsDataURL(file);
    });

    // Reset input to allow same file upload again if needed
    fileInput.value = '';
  });

  clearBtn.addEventListener('click', () => {
    if (confirm('Are you sure you want to clear all uploaded files?')) {
      filesData = [];
      saveToStorage();
      renderFiles();
    }
  });

  // Initial render
  renderFiles();
</script>
