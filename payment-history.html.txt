<script>
  const STORAGE_KEY = 'feeLedgerData';

  let ledgerData = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];

  const paymentBody = document.getElementById('payment-body');
  const searchInput = document.getElementById('search-input');

  // Render payment table rows filtered by search keyword
  function renderPayments(filterText = '') {
    paymentBody.innerHTML = '';

    const filteredData = ledgerData.filter(record => {
      const nameMatch = record.studentName.toLowerCase().includes(filterText.toLowerCase());
      const idMatch = record.studentId.toLowerCase().includes(filterText.toLowerCase());
      return nameMatch || idMatch;
    });

    if (filteredData.length === 0) {
      paymentBody.innerHTML = `<tr><td colspan="6" class="no-data">No payment records found.</td></tr>`;
      return;
    }

    filteredData.forEach((record, index) => {
      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td>${record.studentName}</td>
        <td>${record.studentId}</td>
        <td>${record.paymentDate}</td>
        <td>Rs. ${record.amount.toFixed(2)}</td>
        <td>${record.paymentMode}</td>
        <td>
          <button class="action-btn" onclick="deleteRecord(${index})">Delete</button>
        </td>
      `;
      paymentBody.appendChild(tr);
    });
  }

  // Delete payment record and update storage
  function deleteRecord(index) {
    if (confirm('Are you sure you want to delete this payment record?')) {
      ledgerData.splice(index, 1);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(ledgerData));
      renderPayments(searchInput.value);
    }
  }

  // Event listener for search input
  searchInput.addEventListener('input', () => {
    renderPayments(searchInput.value);
  });

  // Initial render
  renderPayments();

  // Make deleteRecord globally accessible for onclick
  window.deleteRecord = deleteRecord;
</script>
