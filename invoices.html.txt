<script>
  const STORAGE_KEY = 'invoicesData';

  let invoices = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];

  const invoiceBody = document.getElementById('invoice-body');
  const searchInput = document.getElementById('search-input');

  function renderInvoices(filterText = '') {
    invoiceBody.innerHTML = '';

    const filtered = invoices.filter(invoice => {
      const nameMatch = invoice.studentName.toLowerCase().includes(filterText.toLowerCase());
      const invoiceMatch = invoice.invoiceNo.toLowerCase().includes(filterText.toLowerCase());
      return nameMatch || invoiceMatch;
    });

    if (filtered.length === 0) {
      invoiceBody.innerHTML = `<tr><td colspan="7" class="no-data">No invoices found.</td></tr>`;
      return;
    }

    filtered.forEach((inv, idx) => {
      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td>${inv.invoiceNo}</td>
        <td>${inv.studentName}</td>
        <td>${inv.studentId}</td>
        <td>${inv.date}</td>
        <td>Rs. ${inv.amount.toFixed(2)}</td>
        <td class="${inv.status === 'Paid' ? 'status-paid' : 'status-unpaid'}">${inv.status}</td>
        <td>
          <button class="action-btn" onclick="toggleStatus(${idx})">${inv.status === 'Paid' ? 'Mark Unpaid' : 'Mark Paid'}</button>
          <button class="action-btn delete-btn" onclick="deleteInvoice(${idx})">Delete</button>
        </td>
      `;
      invoiceBody.appendChild(tr);
    });
  }

  function toggleStatus(index) {
    invoices[index].status = invoices[index].status === 'Paid' ? 'Unpaid' : 'Paid';
    localStorage.setItem(STORAGE_KEY, JSON.stringify(invoices));
    renderInvoices(searchInput.value);
  }

  function deleteInvoice(index) {
    if (confirm('Delete this invoice?')) {
      invoices.splice(index, 1);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(invoices));
      renderInvoices(searchInput.value);
    }
  }

  searchInput.addEventListener('input', () => {
    renderInvoices(searchInput.value);
  });

  renderInvoices();

  window.toggleStatus = toggleStatus;
  window.deleteInvoice = deleteInvoice;
</script>
