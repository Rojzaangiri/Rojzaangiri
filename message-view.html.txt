<script>
  // The message data will be passed as URL query params (e.g., ?index=0)
  // For demo, we get the index param and fetch message from localStorage

  function getQueryParam(param) {
    const urlParams = new URLSearchParams(window.location.search);
    return urlParams.get(param);
  }

  const STORAGE_KEY = 'noticesData';
  const msgTitleEl = document.getElementById('msgTitle');
  const msgDateEl = document.getElementById('msgDate');
  const msgContentEl = document.getElementById('msgContent');

  function loadMessage() {
    const indexStr = getQueryParam('index');
    if (indexStr === null) {
      msgTitleEl.textContent = "No message selected.";
      return;
    }

    const index = parseInt(indexStr);
    if (isNaN(index)) {
      msgTitleEl.textContent = "Invalid message.";
      return;
    }

    const allMessages = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];

    if (index < 0 || index >= allMessages.length) {
      msgTitleEl.textContent = "Message not found.";
      return;
    }

    const message = allMessages[index];
    msgTitleEl.textContent = message.title;
    msgDateEl.textContent = `Date: ${message.date}`;
    msgContentEl.textContent = message.message;
  }

  loadMessage();
</script>
