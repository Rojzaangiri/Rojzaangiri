<script>
  const form = document.getElementById('exam-form');
  const tableBody = document.querySelector('#exam-table tbody');

  // Load exams from localStorage or empty array
  let exams = JSON.parse(localStorage.getItem('examSchedule')) || [];

  function saveExams() {
    localStorage.setItem('examSchedule', JSON.stringify(exams));
  }

  function renderTable() {
    tableBody.innerHTML = '';
    exams.forEach((exam, index) => {
      const tr = document.createElement('tr');

      tr.innerHTML = `
        <td><input type="text" value="${exam.subject}" data-index="${index}" data-field="subject" /></td>
        <td><input type="date" value="${exam.date}" data-index="${index}" data-field="date" /></td>
        <td><input type="time" value="${exam.start}" data-index="${index}" data-field="start" /></td>
        <td><input type="time" value="${exam.end}" data-index="${index}" data-field="end" /></td>
        <td><button class="action-btn" data-index="${index}">Delete</button></td>
      `;

      tableBody.appendChild(tr);
    });
  }

  function updateExam(index, field, value) {
    exams[index][field] = value;
    saveExams();
    renderTable();
  }

  form.addEventListener('submit', e => {
    e.preventDefault();
    const subject = document.getElementById('subject').value;
    const date = document.getElementById('exam-date').value;
    const start = document.getElementById('start-time').value;
    const end = document.getElementById('end-time').value;

    exams.push({ subject, date, start, end });
    saveExams();
    renderTable();

    form.reset();
  });

  tableBody.addEventListener('input', e => {
    if (e.target.tagName === 'INPUT') {
      const index = e.target.getAttribute('data-index');
      const field = e.target.getAttribute('data-field');
      const value = e.target.value;
      updateExam(index, field, value);
    }
  });

  tableBody.addEventListener('click', e => {
    if (e.target.classList.contains('action-btn')) {
      const index = e.target.getAttribute('data-index');
      exams.splice(index, 1);
      saveExams();
      renderTable();
    }
  });

  // Initial render
  renderTable();
</script>
